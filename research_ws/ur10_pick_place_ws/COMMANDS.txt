────────────────────────────────────────────────────────────────────────────────
STEP 0 — Source ROS2 (run in EVERY terminal before any ros2 command)
────────────────────────────────────────────────────────────────────────────────
source /opt/ros/humble/setup.bash
source ~/research_ws/ur10_pick_place_ws/install/setup.bash

────────────────────────────────────────────────────────────────────────────────
STEP 1 — Build the workspace  (Terminal 1, once after any code changes)
────────────────────────────────────────────────────────────────────────────────
cd ~/research_ws/ur10_pick_place_ws
source /opt/ros/humble/setup.bash
colcon build --cmake-args -DCMAKE_BUILD_TYPE=Release
# After build completes:
source install/setup.bash

────────────────────────────────────────────────────────────────────────────────
STEP 2 — Launch the full simulation  (Terminal 1)
────────────────────────────────────────────────────────────────────────────────
cd ~/research_ws/ur10_pick_place_ws
source /opt/ros/humble/setup.bash && source install/setup.bash

# Default: Ignition GUI + RViz + grasp_executor
ros2 launch ur10_pick_place grasp_pipeline.launch.py

# Options:
#   use_gz_gui:=false          → headless Ignition (faster, no Gazebo window)
#   use_rviz:=false            → skip RViz
#   launch_grasp_executor:=false → skip grasp_executor_node
#   launch_graspgen_bridge:=true → also launch graspgen_bridge_node (needs GraspGen)

# Example: headless Ignition + RViz only (no executor):
ros2 launch ur10_pick_place grasp_pipeline.launch.py \
  use_gz_gui:=false launch_grasp_executor:=false

# Example: full pipeline with GraspGen bridge:
ros2 launch ur10_pick_place grasp_pipeline.launch.py \
  launch_graspgen_bridge:=true

────────────────────────────────────────────────────────────────────────────────
STEP 3 — Verify system is running  (Terminal 2)
────────────────────────────────────────────────────────────────────────────────
source /opt/ros/humble/setup.bash && source ~/research_ws/ur10_pick_place_ws/install/setup.bash

# Check all controllers are active:
ros2 control list_controllers

# Expected output:
#   joint_state_broadcaster[joint_state_broadcaster/JointStateBroadcaster] active
#   joint_trajectory_controller[joint_trajectory_controller/JointTrajectoryController] active
#   gripper_trajectory_controller[joint_trajectory_controller/JointTrajectoryController] active

# Check topics are published:
ros2 topic list | grep -E "(joint_states|graspgen|realsense|oakd)"

# Check move_group is running:
ros2 node list | grep move_group

# Check grasp executor status:
ros2 topic echo /grasp_executor/status --once

# View available camera topics:
ros2 topic list | grep -E "(realsense|oakd)"
#   /realsense_d435/image
#   /realsense_d435/depth_image
#   /realsense_d435/camera_info
#   /realsense_d435/points
#   /oakd_pro/image
#   /oakd_pro/depth_image
#   /oakd_pro/camera_info
#   /oakd_pro/points

────────────────────────────────────────────────────────────────────────────────
STEP 4a — Manual test: trigger a grasp with a fake pose  (Terminal 2)
────────────────────────────────────────────────────────────────────────────────
# Publish a test grasp pose — tin_can approximate location:
ros2 topic pub --once /graspgen/grasp_pose geometry_msgs/msg/PoseStamped \
  "{header: {frame_id: 'base_link'}, \
    pose: {position: {x: 0.55, y: -0.12, z: 1.10}, \
           orientation: {x: 0.0, y: 1.0, z: 0.0, w: 0.0}}}"

# Monitor grasp executor status:
ros2 topic echo /grasp_executor/status

────────────────────────────────────────────────────────────────────────────────
STEP 4b — Run GraspGen inference (requires graspgen_bridge_node running)
────────────────────────────────────────────────────────────────────────────────
# 1. Make sure GraspGen denylist stub exists (prevents cache rebuild):
echo '{}' > /tmp/graspgen_cache/robotiq_3f/denylist_mesh_dis.json

# 2. Launch with GraspGen bridge enabled:
ros2 launch ur10_pick_place grasp_pipeline.launch.py launch_graspgen_bridge:=true

# 3. In another terminal — trigger inference when object is in camera view:
source /opt/ros/humble/setup.bash && source ~/research_ws/ur10_pick_place_ws/install/setup.bash
ros2 topic pub --once /graspgen/trigger std_msgs/msg/Empty "{}"

# 4. Monitor GraspGen status:
ros2 topic echo /graspgen/status

# 5. Monitor grasp executor:
ros2 topic echo /grasp_executor/status

# To use OAK-D Pro instead of RealSense for GraspGen:
ros2 run grasp_executor graspgen_bridge_node \
  --ros-args -r /realsense_d435/points:=/oakd_pro/points

────────────────────────────────────────────────────────────────────────────────
STEP 5 — Check camera streams in RViz
────────────────────────────────────────────────────────────────────────────────
# Add in RViz:
#   Image display → Topic: /realsense_d435/image
#   Image display → Topic: /oakd_pro/image
#   PointCloud2   → Topic: /realsense_d435/points  (frame: realsense_d435_link)
#   PointCloud2   → Topic: /oakd_pro/points          (frame: oakd_pro::camera_link)

# View camera image in terminal:
ros2 run rqt_image_view rqt_image_view

────────────────────────────────────────────────────────────────────────────────
DEBUGGING
────────────────────────────────────────────────────────────────────────────────

# Check TF tree (robot state should be fully connected):
ros2 run tf2_tools view_frames

# Check joint states are publishing:
ros2 topic echo /joint_states

# Check MoveIt planning scene:
ros2 topic echo /monitored_planning_scene --once

# Inspect robot description:
ros2 param get /robot_state_publisher robot_description | head -20

# Check move_group action server:
ros2 action list

# Re-send a grasp pose (after IDLE status):
ros2 topic pub --once /graspgen/grasp_pose geometry_msgs/msg/PoseStamped \
  "{header: {frame_id: 'base_link'}, \
    pose: {position: {x: 0.65, y: 0.1, z: 1.20}, \
           orientation: {x: 0.0, y: 1.0, z: 0.0, w: 0.0}}}"

────────────────────────────────────────────────────────────────────────────────
GRASPGEN TRAINING (if retraining needed)
────────────────────────────────────────────────────────────────────────────────
# Always run this BEFORE training to prevent cache rebuild:
echo '{}' > /tmp/graspgen_cache/robotiq_3f/denylist_mesh_dis.json

cd ~/research_ws/GraspGen

# Generator training:
python3 scripts/train_graspgen.py \
  train.model_name=generator \
  data.root_dir=~/GraspDataGen/graspgen_dataset/splits/robotiq_3f \
  data.object_root_dir=~/GraspDataGen/objects \
  data.grasp_root_dir=~/GraspDataGen/graspgen_dataset/grasp_data/robotiq_3f \
  data.gripper_name=robotiq_3f \
  train.log_dir=~/GraspDataGen/training_logs/robotiq_3f_gen \
  train.num_epochs=500

# On-policy data generation (after generator training):
python3 scripts/generate_onpolicy_data.py --num_grasps 300

# Discriminator retraining (on-policy):
echo '{}' > /tmp/graspgen_cache/robotiq_3f/denylist_mesh_dis.json
python3 scripts/train_graspgen.py \
  train.model_name=discriminator \
  data.load_discriminator_dataset=True \
  data.discriminator_ratio=[0.25,0.20,0.00,0.05,0.0,0.25,0.25] \
  train.checkpoint=~/GraspDataGen/training_logs/robotiq_3f_disc/epoch_500.pth \
  train.log_dir=~/GraspDataGen/training_logs/robotiq_3f_disc_onpolicy \
  train.num_epochs=500
